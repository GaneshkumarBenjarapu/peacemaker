EXERCISE 7 - DOCKER COMPOSE
============================================================

1. Create folder.

mkdir compose-lab
cd compose-lab

2. Create docker-compose.yml with:

services:
  db:
    image: mysql:8.0
    container_name: mysql-container
    restart: always
    environment:
      MYSQL_ROOT_PASSWORD: root123
      MYSQL_DATABASE: wordpress
      MYSQL_USER: wpuser
      MYSQL_PASSWORD: wp123
    volumes:
      - db_data:/var/lib/mysql

  wordpress:
    image: wordpress:latest
    container_name: wordpress-container
    restart: always
    ports:
      - "8083:80"
    environment:
      WORDPRESS_DB_HOST: db:3306
      WORDPRESS_DB_USER: wpuser
      WORDPRESS_DB_PASSWORD: wp123
      WORDPRESS_DB_NAME: wordpress
    depends_on:
      - db

volumes:
  db_data:

3. Run.

docker compose up -d
docker compose ps
docker compose logs
docker volume ls
docker network ls

4. Open http://localhost:8083

5. Stop and remove.

docker compose down
docker compose up -d

6. Remove containers and volumes.

docker compose down -v
